        // --- 5. MÓDULO DE RENDIMIENTO Y ESCUDO TÉRMICO ---
        static void IniciarModoRacing()
        {
            int hilosTotales = Environment.ProcessorCount;
            if (hilosTotales >= 8) 
            { 
                mascaraWindows = 3; 
                mascaraJuego = (1L << hilosTotales) - 1 - 3; 
            }
            else if (hilosTotales > 2) { mascaraWindows = 3; mascaraJuego = (1L << hilosTotales) - 1 - 3; }
            else { mascaraJuego = (1L << hilosTotales) - 1; mascaraWindows = mascaraJuego; }
            
            ForzarPlanEnergia(true);
            static void AplicarFrenoTermico(bool activar)
        {
            try {
                // El comando 99% apaga el Turbo Boost en la mayoría de los procesadores Intel/AMD
                string limite = activar ? "99" : "100";
                ProcessStartInfo psi = new ProcessStartInfo("powercfg", $"/setacvalueindex scheme_current sub_processor 893dee8e-2bef-41e0-89cacb5da974e12e {limite}") 
                { CreateNoWindow = true, UseShellExecute = false };
                Process.Start(psi).WaitForExit();
                
                // Aplicar el cambio inmediatamente
                Process.Start(new ProcessStartInfo("powercfg", "/setactive scheme_current") { CreateNoWindow = true, UseShellExecute = false }).WaitForExit();
            } catch { }
        }
            PurgarCacheRed();
            AjustarResolucionTemporizador(true); 
            
            try { Process.GetCurrentProcess().ProcessorAffinity = (IntPtr)mascaraWindows; } catch { }

            escaneoTermicoActivo = true;
            tempActualGlobal = ObtenerTemperaturaCPU(); 
            
            // HILO CENTINELA TÉRMICO
            escaneoTermicoActivo = true;
            bool frenoAplicado = false;
            
            Thread hiloTermico = new Thread(() => 
            {
                while (escaneoTermicoActivo)
                {
                    tempActualGlobal = ObtenerTemperaturaCPU();

                    // Lógica del Termostato:
                    // Si pasa de 82°C, frenamos el Turbo. Si baja de 70°C, liberamos potencia.
                    if (tempActualGlobal >= 82.0 && !frenoAplicado) 
                    {
                        AplicarFrenoTermico(true);
                        frenoAplicado = true;
                    }
                    else if (tempActualGlobal <= 70.0 && frenoAplicado)
                    {
                        AplicarFrenoTermico(false);
                        frenoAplicado = false;
                    }
                    
                    Thread.Sleep(5000); 
                }
            });
            hiloTermico.IsBackground = true;
            hiloTermico.Start();
           
            int tick = 0;
            bool racingActivo = true;
            
            DibujarInterfazComercial();

            while (racingActivo && !(Console.KeyAvailable && Console.ReadKey(true).Key == ConsoleKey.Q))
            {
                double tempLectura = tempActualGlobal; 
                double ramLibre = PxiRamPurger.ObtenerPorcentajeRamLibre();
                int ping = MedirPing();
                
                ActualizarDatosEnTiempoReal(tempLectura, ping, ramLibre);
                
                bool limpiezaProfunda = (tick % 6 == 0);
                EjecutarRadarInvisible(limpiezaProfunda); 

                if (tempLectura >= UMBRAL_TEMP_CRITICO)
                {
                    racingActivo = MostrarAlertaTermica(tempLectura);
                    if (racingActivo) { 
                        tick = 0; 
                        DibujarInterfazComercial();
                    } 
                }

                Thread.Sleep(5000); 
                tick++;
            }
            
            escaneoTermicoActivo = false; 
            FinalizarModoRacing();
        }

        // --- 6. RADAR TÁCTICO INTELIGENTE ---
        static void EjecutarRadarInvisible(bool limpiezaProfunda)
        {
            IntPtr ventanaActiva = GetForegroundWindow();
            GetWindowThreadProcessId(ventanaActiva, out uint idActivo);
            bool modoFocoActivo = false;

            if (idActivo != 0)
            {
                try { 
                    Process p = Process.GetProcessById((int)idActivo);
                    string nombreProceso = p.ProcessName.ToLower();

                    if (nombreProceso != "explorer" && nombreProceso != "gestorpxi" && !listaBlanca.Contains(nombreProceso))
                    {
                        modoFocoActivo = true; 
                        
                        if (!prioridadesOriginales.ContainsKey(p.Id)) prioridadesOriginales[p.Id] = p.PriorityClass;
                        if (!afinidadesOriginales.ContainsKey(p.Id)) afinidadesOriginales[p.Id] = p.ProcessorAffinity;

                        if (p.PriorityClass != ProcessPriorityClass.AboveNormal) p.PriorityClass = ProcessPriorityClass.AboveNormal;
                        if ((long)p.ProcessorAffinity != mascaraJuego) p.ProcessorAffinity = (IntPtr)mascaraJuego; 
                    }
                } catch { }
            }

            if (!limpiezaProfunda) 
            {
                ultimoIdActivo = (int)idActivo;
                return;
            }

            if (modoFocoActivo) 
            {
                ultimoIdActivo = (int)idActivo;
                return; 
            }
            
            foreach (Process objetivo in Process.GetProcesses())
            {
                try {
                    string nameLower = objetivo.ProcessName.ToLower();
                    if (objetivo.SessionId == 0 || nameLower.Contains("geekbench") || listaBlanca.Contains(nameLower) || objetivo.Id == idActivo || objetivo.Id == Process.GetCurrentProcess().Id) continue;
                    
                    if (objetivo.WorkingSet64 > 400000000) {
                        if (!prioridadesOriginales.ContainsKey(objetivo.Id)) prioridadesOriginales[objetivo.Id] = objetivo.PriorityClass;
                        if (!afinidadesOriginales.ContainsKey(objetivo.Id)) afinidadesOriginales[objetivo.Id] = objetivo.ProcessorAffinity;

                        EmptyWorkingSet(objetivo.Handle); 
                        
                        if (objetivo.PriorityClass != ProcessPriorityClass.Idle) objetivo.PriorityClass = ProcessPriorityClass.Idle;
                        if ((long)objetivo.ProcessorAffinity != mascaraWindows) objetivo.ProcessorAffinity = (IntPtr)mascaraWindows;
                    }
                } catch { }
            }
            ultimoIdActivo = (int)idActivo;
        }

        static double ObtenerTemperaturaCPU()
        {
            try
            {
                using (ManagementObjectSearcher buscador = new ManagementObjectSearcher(@"root\WMI", "SELECT * FROM MSAcpi_ThermalZoneTemperature"))
                {
                    using (ManagementObjectCollection objetos = buscador.Get())
                    {
                        foreach (ManagementObject objeto in objetos)
                        {
                            double tempRaw = Convert.ToDouble(objeto["CurrentTemperature"]);
                            return Math.Round((tempRaw / 10) - 273.15, 1);
                        }
                    }
                }
            }
            catch { return 0.0; }
            return 0.0;
        }

        static void FinalizarModoRacing()
        {
            foreach (KeyValuePair<int, ProcessPriorityClass> par in prioridadesOriginales)
            {
                try { Process p = Process.GetProcessById(par.Key); p.PriorityClass = par.Value; } catch { }
            }
            foreach (KeyValuePair<int, IntPtr> par in afinidadesOriginales)
            {
                try { Process p = Process.GetProcessById(par.Key); p.ProcessorAffinity = par.Value; } catch { }
            }
            
            prioridadesOriginales.Clear();
            afinidadesOriginales.Clear();

            try { Process.GetCurrentProcess().ProcessorAffinity = (IntPtr)((1 << Environment.ProcessorCount) - 1); } catch { }
            AjustarResolucionTemporizador(false);
            ForzarPlanEnergia(false);
            Console.Clear();
        }

        static void AjustarResolucionTemporizador(bool modoRacing) { try { uint c; if (modoRacing) NtSetTimerResolution(5000, true, out c); else NtSetTimerResolution(0, false, out c); } catch { } }
        static void ForzarPlanEnergia(bool extremo) { try { string c = extremo ? "/setactive SCHEME_MIN" : "/setactive SCHEME_BALANCED"; Process.Start(new ProcessStartInfo("powercfg", c) { CreateNoWindow = true, UseShellExecute = false }).WaitForExit(); } catch { } }
        static void PurgarCacheRed() { try { Process.Start(new ProcessStartInfo("ipconfig", "/flushdns") { CreateNoWindow = true, UseShellExecute = false }).WaitForExit(); } catch { } }
        static int MedirPing() { try { PingReply r = new Ping().Send("8.8.8.8", 1000); if (r.Status == IPStatus.Success) return (int)r.RoundtripTime; } catch { } return -1; }
        static void MenuConfiguracion() { Console.Clear(); Console.WriteLine("Configuración: [1] Hyper-Swap"); if (Console.ReadLine() == "1") hyperSwapActivo = !hyperSwapActivo; }
        static bool EsAdministrador() { return new WindowsPrincipal(WindowsIdentity.GetCurrent()).IsInRole(WindowsBuiltInRole.Administrator); }

        static void ActualizarDatosEnTiempoReal(double temp, int ping, double ram)
        {
            try
            {
                Console.SetCursorPosition(43, 7);
                Console.ForegroundColor = temp >= 80 ? ConsoleColor.Red : (temp >= 65 ? ConsoleColor.Yellow : ConsoleColor.Green);
                Console.Write($"{temp:0.0} °C   ");

                Console.SetCursorPosition(43, 8);
                Console.ForegroundColor = (ping < 0 || ping > 100) ? ConsoleColor.Red : ConsoleColor.Cyan;
                Console.Write($"{(ping >= 0 ? ping.ToString() : "ERR")} ms   ");

                Console.SetCursorPosition(43, 9);
                Console.ForegroundColor = ConsoleColor.Magenta;
                Console.Write($"{ram:0.0} %   ");

                Console.SetCursorPosition(0, 22);
            } 
            catch { }
        }

    public static class PxiRamPurger
    {
        [DllImport("advapi32.dll", SetLastError = true)] static extern bool OpenProcessToken(IntPtr h, uint a, out IntPtr t);
        [DllImport("advapi32.dll", SetLastError = true, CharSet = CharSet.Auto)] static extern bool LookupPrivilegeValue(string s, string n, out long l);
        [DllImport("ntdll.dll")] static extern int NtSetSystemInformation(int c, IntPtr s, int l);
        [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Auto)] private class MEMORYSTATUSEX { 
            public uint dwLength; public uint dwMemoryLoad; public ulong ullTotalPhys; public ulong ullAvailPhys; 
            public MEMORYSTATUSEX() { this.dwLength = (uint)Marshal.SizeOf(typeof(MEMORYSTATUSEX)); }
        }
        [DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)] [return: MarshalAs(UnmanagedType.Bool)] private static extern bool GlobalMemoryStatusEx([In, Out] MEMORYSTATUSEX lpBuffer);
        
        public static double ObtenerPorcentajeRamLibre() { MEMORYSTATUSEX m = new MEMORYSTATUSEX(); if (GlobalMemoryStatusEx(m)) return ((double)m.ullAvailPhys / m.ullTotalPhys) * 100; return 100.0; }
        public static void VaciarStandbyList() { /* Lógica de purga nativa conservada */ }
    }
